# 데이터 보호 가이드 - 서비스 기획

## 1. 개요

> **핵심 원칙**: 사용자의 소중한 금연 기록이 절대 손실되지 않도록
> 철저한 안내와 백업 시스템을 제공한다.

---

## 2. 개인정보 보호 안내

### 2.1 핵심 개인정보 보호 공지

> **🔒 여러분의 기록은 다른 곳에 저장되지 않습니다**
>
> Smoking Habit은 사용자의 개인정보를 **절대** 수집하거나
> 외부 서버에 저장하지 않습니다.

```

│                                     │
└─────────────────────────────────────┘
```

### 2.2 개인정보 보호 상세 설명

| 항목 | 내용 |
|------|------|
| **데이터 저장 위치** | 사용자의 기기(브라우저) 내 LocalStorage |
| **서버 전송 여부** | 없음 (서버 자체가 없음) |
| **회원가입 필요** | 불필요 |
| **수집하는 정보** | 없음 |
| **제3자 제공** | 없음 |
| **광고/마케팅 활용** | 없음 |
| **데이터 소유권** | 100% 사용자 본인 |

### 2.4 ⚠️ 데이터가 사라질 수 있는 경우 (중요!)

> **주의**: 데이터가 오직 이 기기에만 저장되기 때문에,
> 아래 상황에서는 금연 기록이 **영구 삭제**될 수 있습니다.


### 8.0 데이터 지속성 강화 전략 요약

| 삭제 위험 요인 | 기술적 대응 방안 | 효과 |
|---------------|-----------------|------|
| 브라우저 데이터 삭제 | IndexedDB + localStorage 이중 저장 | 부분 보호 |
| 시크릿 모드 사용 | 시크릿 모드 감지 → 진입 차단/경고 | 완전 보호 |

### 8.1 다중 저장소 전략 (이중 백업)

```javascript
// 핵심 데이터를 localStorage와 IndexedDB 모두에 저장
// 하나가 삭제되어도 다른 곳에서 복원 가능

const STORAGE_KEY = 'smoking-habit-data';
const DB_NAME = 'SmokingHabitDB';
const STORE_NAME = 'userData';

// IndexedDB 초기화
async function initIndexedDB() {
  return new Promise((resolve, reject) => {
    const request = indexedDB.open(DB_NAME, 1);

    request.onerror = () => reject(request.error);
    request.onsuccess = () => resolve(request.result);

    request.onupgradeneeded = (event) => {
      const db = event.target.result;
      if (!db.objectStoreNames.contains(STORE_NAME)) {
        db.createObjectStore(STORE_NAME, { keyPath: 'id' });
      }
    };
  });
}

// 데이터 저장 (이중 저장)
async function saveData(data) {
  // 1. localStorage에 저장
  try {
    localStorage.setItem(STORAGE_KEY, JSON.stringify(data));
  } catch (e) {
    console.warn('localStorage 저장 실패:', e);
  }

  // 2. IndexedDB에도 저장 (백업)
  try {
    const db = await initIndexedDB();
    const tx = db.transaction(STORE_NAME, 'readwrite');
    const store = tx.objectStore(STORE_NAME);
    await store.put({ id: 'main', data, updatedAt: new Date().toISOString() });
  } catch (e) {
    console.warn('IndexedDB 저장 실패:', e);
  }
}

// 데이터 로드 (복구 로직 포함)
async function loadData() {
  let localData = null;
  let indexedDBData = null;

  // 1. localStorage에서 시도
  try {
    const stored = localStorage.getItem(STORAGE_KEY);
    if (stored) localData = JSON.parse(stored);
  } catch (e) {
    console.warn('localStorage 읽기 실패:', e);
  }

  // 2. IndexedDB에서 시도
  try {
    const db = await initIndexedDB();
    const tx = db.transaction(STORE_NAME, 'readonly');
    const store = tx.objectStore(STORE_NAME);
    const result = await new Promise((resolve, reject) => {
      const request = store.get('main');
      request.onsuccess = () => resolve(request.result);
      request.onerror = () => reject(request.error);
    });
    if (result) indexedDBData = result.data;
  } catch (e) {
    console.warn('IndexedDB 읽기 실패:', e);
  }

  // 3. 복구 로직: 더 최신 데이터 또는 존재하는 데이터 사용
  if (localData && indexedDBData) {
    // 둘 다 있으면 더 최신 것 사용
    const localTime = new Date(localData.updatedAt || 0);
    const idbTime = new Date(indexedDBData.updatedAt || 0);
    return localTime > idbTime ? localData : indexedDBData;
  }

  // 4. 하나만 있으면 그것 사용 + 복구
  if (localData && !indexedDBData) {
    console.log('IndexedDB에서 복구 중...');
    await saveData(localData); // IndexedDB에도 저장
    return localData;
  }

  if (indexedDBData && !localData) {
    console.log('localStorage에서 복구 중...');
    localStorage.setItem(STORAGE_KEY, JSON.stringify(indexedDBData));
    return indexedDBData;
  }

  return null; // 데이터 없음
}
```

### 8.2 Persistent Storage 요청 (삭제 방지)

```javascript
// 브라우저에 "이 데이터 삭제하지 마세요" 요청
// Safari 17+, Chrome, Firefox 지원

async function requestPersistentStorage() {
  if (!navigator.storage || !navigator.storage.persist) {
    console.log('Persistent Storage API 미지원');
    return false;
  }

  // 이미 persistent인지 확인
  const isPersisted = await navigator.storage.persisted();
  if (isPersisted) {
    console.log('이미 Persistent Storage 활성화됨');
    return true;
  }

  // persistent 요청 (사용자 상호작용 후 호출 권장)
  const granted = await navigator.storage.persist();

  if (granted) {
    console.log('Persistent Storage 승인됨 - 데이터 보호 강화');
  } else {
    console.log('Persistent Storage 거부됨 - 기본 모드');
  }

  return granted;
}

// 저장 공간 상태 확인
async function checkStorageStatus() {
  if (!navigator.storage || !navigator.storage.estimate) {
    return null;
  }

  const estimate = await navigator.storage.estimate();
  return {
    usage: estimate.usage,           // 사용 중인 바이트
    quota: estimate.quota,           // 할당된 총 바이트
    usagePercent: ((estimate.usage / estimate.quota) * 100).toFixed(2),
    isPersisted: await navigator.storage.persisted()
  };
}

// 첫 데이터 저장 시 persistent 요청
async function onFirstDataSave() {
  const persisted = await requestPersistentStorage();

  if (!persisted) {
    // persistent 실패 시 사용자에게 PWA 설치 권유
    showPWAInstallPrompt();
  }
}
```

```javascript
  e.preventDefault();
  // 나중에 사용하기 위해 저장

  showInstallButton();

async function promptPWAInstall() {
  if (!deferredPrompt) {
    // iOS는 beforeinstallprompt 미지원 → 수동 안내

      showIOSInstallGuide();

    return;

  const { outcome } = await deferredPrompt.userChoice;

    console.log('PWA 설치 완료!');
    // 설치 후 persistent storage 요청
    await requestPersistentStorage();
```

```javascript
// 백업 파일 생성 (개선된 버전)
async function createBackup() {
  // localStorage + IndexedDB 모두에서 데이터 수집
  const data = await loadData();

  if (!data) {
    alert('백업할 데이터가 없습니다.');
    return;
  }

  const backupData = {
    version: '1.0',
    appName: 'Smoking Habit',
    createdAt: new Date().toISOString(),
    deviceInfo: {
      platform: navigator.platform,
      userAgent: navigator.userAgent.substring(0, 100)
    },
    userData: data
  };

  const blob = new Blob([JSON.stringify(backupData, null, 2)], {
    type: 'application/json'
  });

  const filename = `smoking-habit-backup-${new Date().toISOString().split('T')[0]}.json`;

  // 다운로드 또는 공유
  if (navigator.share && navigator.canShare({ files: [new File([blob], filename)] })) {
    // 모바일: 공유 시트 표시
    try {
      await navigator.share({
        files: [new File([blob], filename, { type: 'application/json' })],
        title: '금연 기록 백업',
        text: '소중한 금연 기록을 안전한 곳에 보관하세요!'
      });
    } catch (e) {
      // 공유 취소 또는 실패 시 다운로드로 폴백
      downloadFile(blob, filename);
    }
  } else {
    // 데스크톱: 직접 다운로드
    downloadFile(blob, filename);
  }

  // 마지막 백업 시간 저장
  localStorage.setItem('last-backup', new Date().toISOString());

  // 성공 알림
  showToast('백업 완료! 안전한 곳에 보관하세요 💾');
}

function downloadFile(blob, filename) {
  const url = URL.createObjectURL(blob);
  const a = document.createElement('a');
  a.href = url;
  a.download = filename;
  a.click();
  URL.revokeObjectURL(url);
}
```

```javascript
// 백업 파일에서 데이터 복원
async function restoreFromBackup(file) {
  try {
    const text = await file.text();
    const backupData = JSON.parse(text);

    // 버전 및 유효성 검사
    if (!backupData.version || !backupData.userData) {
      throw new Error('올바른 백업 파일이 아닙니다.');
    }

    // 기존 데이터와 병합 여부 확인
    const existingData = await loadData();

    if (existingData) {
      const confirmed = confirm(
        `현재 기록이 있습니다.\n\n` +
        `현재: ${existingData.quitDate || '알 수 없음'}부터 금연 중\n` +
        `백업: ${backupData.userData.quitDate || '알 수 없음'}부터 금연 중\n\n` +
        `백업 파일로 덮어쓰시겠습니까?`
      );

      if (!confirmed) return false;
    }

    // 데이터 복원 (이중 저장)
    await saveData(backupData.userData);

    showToast('복원 완료! 금연 기록을 되찾았어요 🎉');

    // 페이지 새로고침하여 반영
    setTimeout(() => location.reload(), 1500);

    return true;
  } catch (e) {
    alert('복원 실패: ' + e.message);
    return false;
  }
}

// 파일 선택 다이얼로그
function showRestoreDialog() {
  const input = document.createElement('input');
  input.type = 'file';
  input.accept = '.json,application/json';

  input.onchange = async (e) => {
    const file = e.target.files[0];
    if (file) {
      await restoreFromBackup(file);
    }
  };

  input.click();
}
```

```javascript
async function initializeApp() {
```
